class CliFailure extends Error {
  constructor(readonly sqlstate: string | null, message: string) {
    super(message);

let queryCounter = 0;
function cli(sqlText: string, label: string): Array<Record<string, unknown>> {
  // The CLI caches the target in <workdir>/supabase/.temp/linked-project.json; it must only ever be the disposable project.
  if (existsSync(join(WORKDIR, "supabase", ".temp", "project-ref"))) throw new Error("the CLI work directory must not be linked (supabase/.temp/project-ref)");
  const cached = join(WORKDIR, "supabase", ".temp", "linked-project.json");
  if (existsSync(cached)) {
    const meta = JSON.parse(readFileSync(cached, "utf8")) as { ref?: string; name?: string };
    if (meta.ref !== REF || meta.name !== PROJECT_NAME) throw new Error("the CLI's cached project is not the disposable rehearsal project");
  }
  mkdirSync(QUERY_DIR, { recursive: true });
  queryCounter += 1;
  const file = join(QUERY_DIR, `${String(queryCounter).padStart(4, "0")}-${label.replace(/[^A-Za-z0-9_-]+/g, "_").slice(0, 60)}.sql`);
  writeFileSync(file, sqlText, "utf8");
  const r = spawnSync(
    process.execPath,
    [CLI_JS, "db", "query", "--linked", "--project-ref", REF, "--workdir", WORKDIR, "--agent", "no", "--output-format", "json", "-f", file],
    { encoding: "utf8", timeout: 150_000, cwd: WORKDIR, windowsHide: true },
  );
  const out = (r.stdout ?? "").trim();
  if (r.status !== 0) {
    let message = out || (r.stderr ?? "").trim() || `CLI exited ${r.status}`;
    try {
      message = (JSON.parse(out) as { error?: { message?: string } }).error?.message ?? message;
    } catch {
      /* keep raw */
    }
    const m = /ERROR:\s+([0-9A-Z]{5}):\s*([\s\S]*?)(?:\\n|"|$)/.exec(message);
    throw new CliFailure(m ? m[1] : null, m ? m[2] : message.slice(0, 500));
  }
  return JSON.parse(out) as Array<Record<string, unknown>>;
}

function lit(value: unknown): string {
  if (value === null || value === undefined) return "NULL";
  if (typeof value === "number") {
    if (!Number.isFinite(value)) throw new Error("non-finite numeric parameter");
    return String(value);
  }
  if (typeof value === "boolean") return value ? "true" : "false";
  if (Array.isArray(value)) return `ARRAY[${value.map(lit).join(", ")}]`;
  if (typeof value === "string") {
    if (value.includes("\u0000")) throw new Error("NUL in parameter");
    return `'${value.replace(/'/g, "''")}'`;
  }
  throw new Error(`unsupported parameter type ${typeof value}`);
}
const inline = (sql: string, params: unknown[]) => sql.replace(/\$(\d+)/g, (_m, n: string) => lit(params[Number(n) - 1]));

function serverNow(): string {
  return String(one<{ now: string }>(read(`select ${utc("pg_catalog.clock_timestamp()")} as now`, "now")).now);
}

function verifyOnce(): VerifyRun {
  const at = serverNow();
  try {
    cli(SQL.m2cVerify, "m2c-verify");
    return { at, passed: true, message: "no exception: M2C verification PASSED", via: "management-api" };
  } catch (error) {
    return { at, passed: false, message: (error as Error).message, via: "management-api" };
  }
}

async function verifyDoublePass(runs: VerifyRun[], attempts = 1): Promise<boolean> {
  let first: VerifyRun | null = null;
  for (let i = 0; i < attempts; i += 1) {
    first = verifyOnce();
    runs.push(first);
    if (first.passed) break;
    await sleep(1_000);
  }
  if (!first?.passed) return false;
  await sleep(DOUBLE_PASS_GAP_MS);
  const second = verifyOnce();
  runs.push(second);
  return second.passed && Date.parse(second.at) - Date.parse(first.at) >= DOUBLE_PASS_GAP_MS;
}

  return ok;
}

function expectRecorded(ctx: EvalCtx, label: string, row: ResultRow | undefined) {
  ctx.check(`${label}: V2 call returned`, row?.ok === true, row ? { sqlstate: row.sqlstate, message: row.message } : "no result");
  const v = row!.value;
  ctx.check(`${label}: recorded through the existing mapping under the world's attestation generations`,
    v?.status === "recorded" && v?.fixture_resolution === "existing_mapping" && v?.fixture_id === ctx.w.fixtureId &&
      v?.attestation_generations?.league === ctx.w.generations.league &&
      v?.attestation_generations?.team_home === ctx.w.generations.team_home &&
      v?.attestation_generations?.team_away === ctx.w.generations.team_away,
    v);
  return v;
}

}

function chooseRunTag(): string {
  for (let attempt = 0; attempt < 8; attempt += 1) {
    const tag = String(randomInt(100_000, 1_000_000));
    const used = one<{ n: number }>(read(inline(
      `select ((select count(*) from public.provider_team_mappings where provider_team_id like $1)
             + (select count(*) from public.provider_competition_mappings where provider_competition_id like $1)
             + (select count(*) from public.provider_fixture_mappings where provider_fixture_id like $1)
             + (select count(*) from public.provider_player_mappings where provider_player_id like $2))::int as n`,
      [`20${tag}%`, `1${tag}%`]), "run-tag"));
    if (Number(used.n) === 0) return tag;
  }
  throw new Error("could not choose an unused synthetic id range");
}

function install(): void {
  const texts: Array<[string, string]> = [
    ["m2c_revoke", SQL.m2cRevoke],
    ["m2c_verify", SQL.m2cVerify],
    ["lock_snapshot", LOCK_SNAPSHOT_SQL],
    ["sig:v2", V2],
    ["sig:v1", V1],
    ["observe:team_trust", SQL.teamTrust],
    ["observe:impact", SQL.impact],
    ["observe:trusted_read", SQL.trustedRead],
    ["observe:items", SQL.items],
    ["observe:footprint", SQL.footprint],
    ["observe:order", SQL.order],
  ];
  const inserts = texts.map(([name, body]) => `insert into kapiloto_rehearsal.sqltext (name, body) values (${lit(name)}, ${lit(body)});`).join("\n");
  write(`${SQL.install}\n${inserts}\nselect count(*)::int as n from kapiloto_rehearsal.sqltext;`, "install");
  const check = read("select (select body from kapiloto_rehearsal.sqltext where name = 'm2c_revoke') as revoke, (select body from kapiloto_rehearsal.sqltext where name = 'm2c_verify') as verify", "install-check");
  const row = one<{ revoke: string; verify: string }>(check);
  if (row.revoke !== SQL.m2cRevoke || row.verify !== SQL.m2cVerify) throw new Error("installed M2C text differs from docs/operations/sql");
}

async function runExperiment(spec: Spec, runTag: string, index: number): Promise<Experiment> {
  const ex: Experiment = { id: spec.id, title: spec.title, outcome: "fail", sessions: {}, handshakes: [], verify_runs: [], assertions: [] };
  evidence.experiments.push(ex);
  const check = checker(ex);
  try {
    log(`${spec.id}: creating synthetic world`);
    const w = await createRehearsalWorld(worldSession, runTag, index, spec.id);
    ex.world = { provider_fixture_id: w.providerFixtureId, run_id: w.runId };
    const params = { fixture_id: w.fixtureId, run_id: w.runId, provider_home_team_id: w.providerHomeTeamId };
    const keys = { identity: "kapiloto_v4_provider_identity:sportmonks", pair: w.pairKey, fixture: w.fixtureKey };
    const revocation = {
      provider: "sportmonks",
      provider_team_id: w.providerHomeTeamId,
      expected_team_id: w.homeTeamId,
      reason: "synthetic real-PostgreSQL rehearsal revocation",
      revoked_by: "synthetic rehearsal operator",
    };
    if (!harnessSource.includes('reason: "synthetic real-PostgreSQL rehearsal revocation"')) throw new Error("harness revocation payload changed");
    const scriptRows: string[] = [];
    for (const [actor, ops] of Object.entries({ ...spec.actors, O: spec.driver })) {
      ops.forEach((op, i) => scriptRows.push(`(${lit(spec.id)}, ${lit(actor)}, ${i + 1}, ${lit(op.op)}, ${lit(JSON.stringify(op.arg ?? {}))}::jsonb)`));
    }
    write(
      `insert into kapiloto_rehearsal.world (exp, payload, revocation, keys, params) values (${lit(spec.id)}, ${lit(JSON.stringify(w.payload))}::jsonb, ${lit(JSON.stringify(revocation))}::jsonb, ${lit(JSON.stringify(keys))}::jsonb, ${lit(JSON.stringify(params))}::jsonb);
       insert into kapiloto_rehearsal.script (exp, actor, seq, op, arg) values ${scriptRows.join(",\n")};
       select 1 as ok;`,
      `${spec.id}-world-script`,
    );

    write(`${SIMULATED_M2B_GRANT};\nselect 1 as ok;`, `${spec.id}-simulated-grant`);
    const grant = privileges();
    check("simulated M2B grant opened V2 to service_role only; V1 stays revoked", grant.v2 && !grant.v1, grant);

    // Every backend starts at the same minute boundary (pg_cron, GMT).
    const nowEpoch = Number(one<{ e: number }>(read("select extract(epoch from pg_catalog.clock_timestamp())::float8 as e", "epoch")).e);
    const start = new Date(Math.ceil((nowEpoch + 20) / 60) * 60 * 1000);
    const cronExpr = `${start.getUTCMinutes()} ${start.getUTCHours()} ${start.getUTCDate()} ${start.getUTCMonth() + 1} *`;
    const jobs: Array<[string, string]> = [[`kapiloto-rehearsal-${runTag}-${spec.id}-O`, `select kapiloto_rehearsal.driver(${lit(runTag)}, ${lit(spec.id)})`]];
    for (const actor of Object.keys(spec.actors)) jobs.push([`kapiloto-rehearsal-${runTag}-${spec.id}-${actor}`, `select kapiloto_rehearsal.actor(${lit(runTag)}, ${lit(spec.id)}, ${lit(actor)})`]);
    const scheduled = one<{ jobs: Array<{ name: string; jobid: number }> }>(write(
      `select json_agg(json_build_object('name', v.n, 'jobid', cron.schedule(v.n, ${lit(cronExpr)}, v.c)) order by v.n) as jobs
       from (values ${jobs.map(([n, c]) => `(${lit(n)}, ${lit(c)})`).join(", ")}) as v(n, c);`,
      `${spec.id}-schedule`,
    ));
    ex.schedule = { cron: `${cronExpr} (GMT; start ${start.toISOString()})`, jobs: scheduled.jobs };
    const jobIds = scheduled.jobs.map((j) => Number(j.jobid));
    log(`${spec.id}: ${jobIds.length} one-shot pg_cron jobs scheduled for ${start.toISOString()}`);

    // Wait for the driver and every actor to commit (bounded).
    const deadline = start.getTime() + 120_000;
    let status: { done: number; ended: number } = { done: 0, ended: 0 };
    while (Date.now() < deadline) {
      await sleep(2_000);
      status = one<{ done: number; ended: number }>(read(
        `select (select count(*) from kapiloto_rehearsal.evidence where run_tag = ${lit(runTag)} and exp = ${lit(spec.id)} and kind = 'done')::int as done,
                (select count(distinct actor) from kapiloto_rehearsal.result where run_tag = ${lit(runTag)} and exp = ${lit(spec.id)} and op = 'end')::int as ended`,
        `${spec.id}-poll`));
      if (Number(status.done) === 1 && Number(status.ended) === Object.keys(spec.actors).length) break;
    }
    const unscheduled = one<{ n: number }>(write(
      `select count(cron.unschedule(jobid))::int as n from cron.job where jobname like ${lit(`kapiloto-rehearsal-${runTag}-${spec.id}-%`)};`, `${spec.id}-unschedule`));
    log(`${spec.id}: driver done=${status.done}, actors ended=${status.ended}/${Object.keys(spec.actors).length}; unscheduled ${unscheduled.n}`);

    const collected = one<{ c: { evidence: EvRow[] | null; results: ResultRow[] | null; runs: unknown[] | null } }>(read(
      `select json_build_object(
         'evidence', (select json_agg(e order by e.seq) from kapiloto_rehearsal.evidence e where e.run_tag = ${lit(runTag)} and e.exp = ${lit(spec.id)}),
         'results', (select json_agg(r order by r.actor, r.seq) from kapiloto_rehearsal.result r where r.run_tag = ${lit(runTag)} and r.exp = ${lit(spec.id)}),
         'runs', (select json_agg(json_build_object('jobid', d.jobid, 'runid', d.runid, 'job_pid', d.job_pid, 'status', d.status, 'return_message', d.return_message,
                    'start_time', ${utc("d.start_time")}, 'end_time', ${utc("d.end_time")}) order by d.jobid)
                  from cron.job_run_details d where d.jobid = any(${lit(jobIds)}::bigint[]))
       ) as c`, `${spec.id}-collect`)).c;
    const evRows = collected.evidence ?? [];
    const resRows = collected.results ?? [];
    ex.driver_evidence = evRows;
    ex.actor_results = resRows;
    ex.cron_runs = collected.runs;

    const done = evRows.find((e) => e.kind === "done");
    check("the driver (observer) completed its program", done?.ok === true, done?.label ?? "driver did not finish");
    const sessionsEv = evRows.find((e) => e.kind === "sessions");
    const pids = (sessionsEv?.data?.pids ?? {}) as Record<string, number>;
    ex.sessions = pids;
    const pidList = Object.values(pids).map(Number);
    check("every session is a distinct PostgreSQL backend", pidList.length === Object.keys(spec.actors).length + 1 && new Set(pidList).size === pidList.length, pids);
    const activity = (sessionsEv?.data?.activity ?? []) as Array<{ pid: number; application_name: string }>;
    check("pg_stat_activity shows each session pid with its own application_name",
      Object.entries(pids).every(([name, pid]) => activity.some((a) => a.pid === pid && a.application_name === `${APPLICATION_NAME_PREFIX}${runTag}:${spec.id}-${name}`)), activity);
    const runs = (collected.runs ?? []) as Array<{ jobid: number; job_pid: number; status: string }>;
    const jobPid = new Map(scheduled.jobs.map((j) => [j.name.split("-").pop() as string, runs.find((r) => Number(r.jobid) === Number(j.jobid))?.job_pid]));
    check("pg_cron job_run_details independently records each session's backend pid (one run per job, all succeeded)",
      runs.length === jobIds.length && runs.every((r) => r.status === "succeeded") && Object.entries(pids).every(([name, pid]) => Number(jobPid.get(name)) === Number(pid)), runs);
    for (const e of evRows.filter((r) => r.kind === "handshake")) ex.handshakes.push({ label: e.label, at: e.at, ...(e.data ?? {}) });
    for (const e of evRows.filter((r) => r.kind === "verify")) ex.verify_runs.push({ at: e.data?.at, passed: e.data?.passed, message: e.data?.message, via: "driver backend (pg_cron)" });
    const resultsOf = (actor: string, op: string) => resRows.filter((r) => r.actor === actor && r.op === op);
    for (const actor of Object.keys(spec.actors)) {
      const end = resRows.find((r) => r.actor === actor && r.op === "end");
      const gates = resRows.filter((r) => r.actor === actor && (r.op === "gate" || r.op === "role" || r.op === "reset_role" || r.op === "hold"));
      check(`${actor}: ran its whole script (gates, role switches and lock holds succeeded)`, end?.ok === true && gates.every((g) => g.ok === true), resRows.filter((r) => r.actor === actor).map((r) => ({ op: r.op, ok: r.ok, sqlstate: r.sqlstate, message: r.message })));
    }
    spec.evaluate({
      check,
      w,
      pids,
      result: (actor, op) => resultsOf(actor, op)[0],
      results: resultsOf,
      observed: (name) => evRows.filter((e) => e.kind === "observe" && e.label === name).map((e) => (e.data ?? []) as any[]),
      handshake: (labelStart) => evRows.find((e) => e.kind === "handshake" && e.label.startsWith(labelStart)),
      ev: (kind) => evRows.filter((e) => e.kind === kind),
    });

    if (spec.m2c) {
      // The actor wrapper is the only object referencing V2: drop it, then prove drain twice.
      write("drop function kapiloto_rehearsal.actor(text, text, text);\nselect 1 as ok;", `${spec.id}-drop-actor`);
      const passed = await verifyDoublePass(ex.verify_runs);
      check("M2C verification PASSES twice at least the gap apart after drain (fallback actor wrapper dropped)", passed, ex.verify_runs);
      write(`${ACTOR_FUNCTION_DDL}\nrevoke all on function kapiloto_rehearsal.actor(text, text, text) from public;\nselect 1 as ok;`, `${spec.id}-recreate-actor`);
    }
    ex.outcome = "pass";
  } catch (error) {
    if (error instanceof InconclusiveRun) {
      ex.outcome = "inconclusive";
      ex.error = { kind: "inconclusive", message: error.message };
    } else if (error instanceof AssertionFailure) {
      ex.outcome = "fail";
      ex.error = { kind: "assertion_failed", message: error.message, observed: error.observed };
    } else {
      ex.outcome = "fail";
      ex.error = { kind: "error", message: (error as Error).message };
    }
  }
  log(`${spec.id}: ${ex.outcome}${ex.error ? ` (${ex.error.message})` : ""}`);
  return ex;
}

async function cleanup(runTag: string | null): Promise<void> {
  const c = evidence.cleanup;
  c.ran = true;
  try {
    if (runTag) {
      const t = read(inline(
        `select pid from pg_catalog.pg_stat_activity where application_name like $1 and pid <> pg_catalog.pg_backend_pid() and pg_catalog.pg_terminate_backend(pid)`,
        [`${APPLICATION_NAME_PREFIX}${runTag}:%`]), "cleanup-terminate");
      c.terminated_pids = t.map((r) => Number(r.pid));
      const u = one<{ n: number }>(read(`select count(cron.unschedule(jobid))::int as n from cron.job where jobname like ${lit(`kapiloto-rehearsal-${runTag}-%`)}`, "cleanup-unschedule"));
      c.unscheduled_jobs = Number(u.n);
    }
    read("set lock_timeout = '10s';\ndrop schema if exists kapiloto_rehearsal cascade;\nselect 1 as ok;", "cleanup-drop-schema");
    c.fallback_objects_dropped = true;
    cli(SQL.m2cRevoke, "cleanup-m2c-revoke-body");
    c.m2c_body_executed = true;
    const passed = await verifyDoublePass(c.verify_runs, 10);
    const acl = privileges();
    c.final_privileges = { v2_service_role: acl.v2, v1_service_role: acl.v1 };
    c.leftover_cron_jobs = Number(one<{ n: number }>(read("select count(*)::int as n from cron.job", "cleanup-cron-count")).n);
    if (!passed || acl.v2 || acl.v1) c.error = "final M2C verification did not PASS twice; run docs/operations/sql/kapiloto-v4-m2c-revoke-body.sql and the verification manually on the disposable project";
  } catch (error) {
    c.error = `cleanup failed (${(error as Error).message}); run docs/operations/sql/kapiloto-v4-m2c-revoke-body.sql and the verification manually on the disposable project`;
  }
}

async function main(): Promise<number> {
  let runTag: string | null = null;
  let wrote = false;
  try {
    log("guards (read-only)");
    if (!runGuards()) {
      evidence.outcome = "refused";
      log(`refused: ${evidence.guards.filter((g) => !g.passed).map((g) => g.name).join(" | ")}`);
      return 2;
    }
    if (process.argv.includes("--guards-only")) {
      log(`guards passed (${evidence.guards.length}); --guards-only: nothing written`);
      process.stdout.write(`${JSON.stringify({ guards: evidence.guards.map((g) => [g.passed, g.name]), function_identity: evidence.function_identity.map((f: any) => [f.matches_checkout, f.signature]), sql_sources: evidence.sql_sources }, null, 1)}\n`);
      return 0;
    }
    runTag = chooseRunTag();
    evidence.run_tag = runTag;
    log(`run tag ${runTag}; installing fallback objects`);
    wrote = true;
    install();
    let outcome: "pass" | "fail" | "inconclusive" = "pass";
    for (const [index, spec] of SPECS.entries()) {
      const ex = await runExperiment(spec, runTag, index);
      if (ex.outcome !== "pass") {
        outcome = ex.outcome;
        break;
      }
    }
    evidence.outcome = outcome;
  } catch (error) {
    evidence.outcome = "fail";
    evidence.error = { kind: "error", message: (error as Error).message };
    log(`error: ${(error as Error).message}`);
  } finally {
    if (wrote) {
      log("cleanup: terminate, unschedule, drop fallback objects, M2C body, double verify");
      await cleanup(runTag);
      if (evidence.cleanup.error) {
        evidence.outcome = "fail";
        log(`cleanup error: ${evidence.cleanup.error}`);
      }
    }
    evidence.finished_at = new Date().toISOString();
    const secrets = [REF, PRODUCTION_REF, `db.${REF}.supabase.co`];
    const redacted = redactEvidence(evidence, secrets);
    const serialized = `${JSON.stringify(redacted, null, 2)}\n`;
    assertEvidenceRedacted(serialized, secrets);
    mkdirSync(dirname(OUT), { recursive: true });
    writeFileSync(OUT, serialized, "utf8");
    log(`evidence ${OUT} sha256=${sha256(serialized)} outcome=${evidence.outcome}`);
  }
  return evidence.outcome === "pass" ? 0 : 1;
}

main().then((code) => {
  process.exitCode = code;
});
